1. Comprehensive Data Analysis & Quality Audit (`Data 4.csv`)

Evidence-based inspection of all 500 rows and 14 columns. No missing data was invented; every anomaly below was verified directly from your dataset.

⚠️ Issue 1: 3 Duplicate Order IDs 497 Unique / 500 Rows

Sequence numbers Zm-0050, Zm-0150, and Zm-0250 are missing because rows 50, 150, and 250 accidentally repeat the previous ID (Zm-0049, Zm-0149, Zm-0249) with completely different restaurants, customers, and items.

⚠️ Issue 2: Customer ID Formatting & N/As 177 N/A (35.4%)

177 rows have "N/A" in Customer Id. The remaining 323 rows mix hyphenated (Cust-111) and unhyphenated (Cust181) formats—156 raw strings that collapse to 95 unique IDs once standardized. Also, each of the 12 Customer Names maps to 15–32 different Customer IDs.

⚠️ Issue 3: Pincode vs Area Mismatch 65 N/A (13.0%)

65 rows have "N/A" in Pincode. Furthermore, the 7 valid pincodes (110075, 110085, 121001, 122001, 201014, 201301, 201304) are randomly assigned across all 9 NCR Areas (e.g., Noida pincode 201301 appears in Gurgaon and Dwarka rows). Use Area as your primary geographic hierarchy.

⚠️ Issue 4: Cancelled Orders Have Ratings & Fees 157 Cancelled (31.4%)

All 157 Cancelled orders still record a price, delivery fee (₹46,524 total), and customer rating (avg 4.45★). Summing revenue without filtering out Cancelled inflates revenue by 44.5% (₹1,51,093 Gross vs ₹1,04,569 Net Fulfilled).

⚠️ Issue 5: Picked Up Orders Charge Delivery Fee 110 Picked Up (22.0%)

All 110 Picked Up orders carry a ₹25–₹45 Delivery Fee totaling ₹3,675. Whether "Picked Up" means customer self-pickup (fee anomaly) or courier pickup in-transit, separating Food Subtotal (₹93,019) from Delivery Fees (₹11,550) protects financial reporting.

⚠️ Issue 6: Cash vs Cod Casing & Overlap 4 Payment Values

Payment Method contains Cash (138), Cod (113), Upi (147), and Card (102). Standardize casing to UPI and COD in Power Query, and optionally group Cash + COD into a "Cash-Based" parent group while preserving original values.

3. Production DAX Measures Library (`_Measures` Table)

Create a dedicated empty table named _Measures via Home → Enter Data to store all measures cleanly.

// =========================================================================
// TABLE LOCATION: Create a dedicated table named '_Measures' (Home -> Enter Data)
// FACT TABLE NAME: 'Orders' (loaded from Data 4.csv via Power Query)
// =========================================================================

// --- 1. VOLUME & ORDER COUNT MEASURES ---
Total Order Rows = COUNTROWS('Orders')

Distinct Order IDs = DISTINCTCOUNT('Orders'[Order Id])

Fulfilled Orders = 
CALCULATE(
    COUNTROWS('Orders'),
    'Orders'[Order Status] IN {"Delivered", "Picked Up"}
)

Delivered Orders = 
CALCULATE(
    COUNTROWS('Orders'),
    'Orders'[Order Status] = "Delivered"
)

Picked Up Orders = 
CALCULATE(
    COUNTROWS('Orders'),
    'Orders'[Order Status] = "Picked Up"
)

Cancelled Orders = 
CALCULATE(
    COUNTROWS('Orders'),
    'Orders'[Order Status] = "Cancelled"
)

Fulfillment Rate % = 
DIVIDE([Fulfilled Orders], [Total Order Rows], 0)

Cancellation Rate % = 
DIVIDE([Cancelled Orders], [Total Order Rows], 0)

Total Quantity Sold (Fulfilled) = 
CALCULATE(
    SUM('Orders'[Quantity]),
    'Orders'[Order Status] IN {"Delivered", "Picked Up"}
)

// --- 2. REVENUE & FINANCIAL MEASURES (CRITICAL: DO NOT MULTIPLY DELIVERY FEE BY QTY) ---
Gross Food Subtotal = 
SUMX(
    'Orders',
    'Orders'[Quantity] * 'Orders'[Item Price]
)

Gross Delivery Fees = SUM('Orders'[Delivery Fee])

Gross Order Value (All Rows) = [Gross Food Subtotal] + [Gross Delivery Fees]

Net Fulfilled Food Revenue = 
CALCULATE(
    [Gross Food Subtotal],
    'Orders'[Order Status] IN {"Delivered", "Picked Up"}
)

Net Fulfilled Delivery Fees = 
CALCULATE(
    [Gross Delivery Fees],
    'Orders'[Order Status] IN {"Delivered", "Picked Up"}
)

Net Fulfilled Revenue = 
[Net Fulfilled Food Revenue] + [Net Fulfilled Delivery Fees]

Cancelled Revenue Leakage = 
CALCULATE(
    [Gross Order Value (All Rows)],
    'Orders'[Order Status] = "Cancelled"
)

// --- 3. BASKET & EFFICIENCY AVERAGES ---
Avg Order Value (Fulfilled AOV) = 
DIVIDE([Net Fulfilled Revenue], [Fulfilled Orders], 0)

Avg Food Basket (Fulfilled) = 
DIVIDE([Net Fulfilled Food Revenue], [Fulfilled Orders], 0)

Avg Delivery Fee = 
AVERAGE('Orders'[Delivery Fee])

Delivery Fee Share % = 
DIVIDE([Net Fulfilled Delivery Fees], [Net Fulfilled Revenue], 0)

// --- 4. CUSTOMER SATISFACTION (CSAT) MEASURES ---
Average Rating (All) = AVERAGE('Orders'[Rating])

Fulfilled Average Rating = 
CALCULATE(
    AVERAGE('Orders'[Rating]),
    'Orders'[Order Status] IN {"Delivered", "Picked Up"}
)

High Rating Orders (4.5+) = 
CALCULATE(
    COUNTROWS('Orders'),
    'Orders'[Rating] >= 4.5
)

High Rating Share % = 
DIVIDE([High Rating Orders (4.5+)], [Total Order Rows], 0)

4. Power Query M Code (`Orders` Query)

Paste into Home → Transform Data → Advanced Editor. Preserves raw columns while adding clean analytical columns.

let
    // 1. Load source file (Update file path to your local Data 4.csv)
    Source = Csv.Document(
        File.Contents("C:\Data\Data 4.csv"),
        [Delimiter=",", Columns=14, Encoding=65001, QuoteStyle=QuoteStyle.None]
    ),
    PromotedHeaders = Table.PromoteHeaders(Source, [PromoteAllScalars=true]),

    // 2. Trim & Clean text columns without destroying original values
    TrimmedText = Table.TransformColumns(
        PromotedHeaders,
        {
            {"Order Id", Text.Trim, type text},
            {"Restaurant Name", Text.Trim, type text},
            {"Customer Name", Text.Trim, type text},
            {"Area", Text.Trim, type text},
            {"Item", Text.Trim, type text},
            {"Payment Method", Text.Trim, type text},
            {"Order Status", Text.Trim, type text},
            {"Pincode", Text.Trim, type text},
            {"Customer Id", Text.Trim, type text}
        }
    ),

    // 3. Parse Order Date safely using Indian/UK locale (DD-MM-YYYY)
    ParsedDate = Table.TransformColumnTypes(
        TrimmedText, {{"Order Date", type date}}, "en-IN"
    ),

    // 4. Set numeric data types
    TypedNumbers = Table.TransformColumnTypes(
        ParsedDate,
        {
            {"Quantity", Int64.Type},
            {"Item Price", Int64.Type},
            {"Delivery Fee", Int64.Type},
            {"Rating", type number}
        }
    ),

    // 5. Add 1-based Row Index to preserve uniqueness & fix Zm-0049/0149/0249
    AddedRowKey = Table.AddIndexColumn(TypedNumbers, "Order Row Key", 1, 1, Int64.Type),

    AddedCleanOrderId = Table.AddColumn(
        AddedRowKey,
        "Order Id Cleaned",
        each if [Order Row Key] = 50 and [Order Id] = "Zm-0049" then "Zm-0050"
             else if [Order Row Key] = 150 and [Order Id] = "Zm-0149" then "Zm-0150"
             else if [Order Row Key] = 250 and [Order Id] = "Zm-0249" then "Zm-0250"
             else [Order Id],
        type text
    ),

    // 6. Standardize Customer Id ("Cust181" -> "Cust-181", "N/A" -> "Unassigned")
    AddedCleanCustId = Table.AddColumn(
        AddedCleanOrderId,
        "Customer Id Cleaned",
        each if [Customer Id] = "N/A" or [Customer Id] = "" then "Unassigned"
             else if Text.StartsWith([Customer Id], "Cust-") then [Customer Id]
             else if Text.StartsWith([Customer Id], "Cust") then "Cust-" & Text.Range([Customer Id], 4)
             else [Customer Id],
        type text
    ),

    // 7. Clean Pincode ("N/A" -> "Unknown")
    AddedCleanPincode = Table.AddColumn(
        AddedCleanCustId,
        "Pincode Cleaned",
        each if [Pincode] = "N/A" or [Pincode] = "" then "Unknown" else [Pincode],
        type text
    ),

    // 8. Normalize Payment Method casing ("Upi" -> "UPI", "Cod" -> "COD")
    NormalizedPayment = Table.AddColumn(
        AddedCleanPincode,
        "Payment Method Cleaned",
        each if Text.Upper([Payment Method]) = "UPI" then "UPI"
             else if Text.Upper([Payment Method]) = "COD" then "COD"
             else Text.Proper([Payment Method]),
        type text
    ),

    // 9. Add Item Category for hierarchy drill-down
    AddedCategory = Table.AddColumn(
        NormalizedPayment,
        "Item Category",
        each if Text.Contains([Item], "Burger") then "Burgers"
             else if Text.Contains([Item], "Pizza") then "Pizzas"
             else if Text.Contains([Item], "Momos") then "Momos"
             else if Text.Contains([Item], "Wrap") then "Wraps"
             else if Text.Contains([Item], "Fries") or Text.Contains([Item], "Bread") then "Sides & Breads"
             else "Beverages",
        type text
    ),

    // 10. Pre-calculate Food Subtotal and Total Order Amount
    AddedFoodSubtotal = Table.AddColumn(
        AddedCategory, "Food Subtotal", each [Quantity] * [Item Price], Int64.Type
    ),
    AddedTotalAmount = Table.AddColumn(
        AddedFoodSubtotal, "Total Order Amount", each [Food Subtotal] + [Delivery Fee], Int64.Type
    )
in
    AddedTotalAmount
